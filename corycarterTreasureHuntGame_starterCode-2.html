The theme of this project is a popular treasure hunt game in which the player needs to find the treasure before the pirate does. While you will not be developing the entire game, you will write the part of the game that represents the intelligent agent, which is a pirate in this case. The pirate will try to find the optimal path to the treasure using deep Q-learning.

The second class, GameExperience.py, stores the episodes – that is, all the states that come in between the initial state and the terminal state. This is later used by the agent for learning by experience, called "exploration". This notebook shows how to play a game. def show(qmaze):
    plt.grid('on')
    nrows, ncols = qmaze.maze.shape
    ax = plt.gca()
    ax.set_xticks(np.arange(0.5, nrows, 1))
    ax.set_yticks(np.arange(0.5, ncols, 1))
    ax.set_xticklabels([])
    ax.set_yticklabels([])
    canvas = np.copy(qmaze.maze)
    for row,col in qmaze.visited:
        canvas[row,col] = 0.6
    pirate_row, pirate_col, _ = qmaze.state
    canvas[pirate_row, pirate_col] = 0.3   # pirate cell
    canvas[nrows-1, ncols-1] = 0.9 # treasure cell
    img = plt.imshow(canvas, interpolation='none', cmap='gray')
    return img

def play_game(model, qmaze, pirate_cell, max_steps=None):
    qmaze.reset(pirate_cell)
    envstate = qmaze.observe()
    steps = 0
    if max_steps is None:
        max_steps = qmaze.maze.size * 4  # safety cutoff

    while steps < max_steps:
        state = np.asarray(envstate, dtype=np.float32)
        if state.ndim == 1:
            state = np.expand_dims(state, axis=0)

        q_values = model(state, training=False).numpy()
        action = np.argmax(q_values[0])

        envstate, reward, game_status = qmaze.act(action)
        steps += 1

        if game_status == 'win':
            return True
        elif game_status == 'lose':
            return False

    return False  # timed out with no result

def completion_check(model, maze_or_qmaze, max_steps=None):
    # Accept either raw numpy maze or TreasureMaze instance
    if isinstance(maze_or_qmaze, TreasureMaze):
        qmaze = maze_or_qmaze
    else:
        qmaze = TreasureMaze(maze_or_qmaze)

    for cell in qmaze.free_cells:
        if not qmaze.valid_actions(cell):
            continue
        if not play_game(model, qmaze, cell, max_steps=max_steps):
            return False
    return True

def build_model(maze):
    model = Sequential()
    model.add(Dense(maze.size, input_shape=(maze.size,)))
    model.add(PReLU())
    model.add(Dense(maze.size))
    model.add(PReLU())
    model.add(Dense(num_actions))
    model.compile(optimizer='adam', loss='mse')
    return model

loss_fn = tf.keras.losses.MeanSquaredError()
optimizer = tf.keras.optimizers.Adam()

@tf.function
def train_step(x, y):
    with tf.GradientTape() as tape:
        q_values = model(x, training=True)
        loss = loss_fn(y, q_values)
    grads = tape.gradient(loss, model.trainable_variables)
    optimizer.apply_gradients(zip(grads, model.trainable_variables))
    return loss

def qtrain(model, maze, **opt): # Use the global epsilon settings (defined earlier in the notebook) global epsilon

# Hyperparameters
n_epoch = opt.get('n_epoch', 15000)          # how many epochs to train
max_memory = opt.get('max_memory', 1000)     # replay memory size
data_size = opt.get('data_size', 50)         # batch size for training
target_update_freq = opt.get('target_update_freq', 50)  # how often to sync target network

# Start time (for printing how long training took)
start_time = datetime.datetime.now()

# Build the environment from the maze
qmaze = TreasureMaze(maze)

# Target network to stabilize learning
target_model = clone_model(model)
target_model.set_weights(model.get_weights())

# Experience replay object (uses both current model and target model)
experience = GameExperience(model, target_model, max_memory=max_memory)

# Tracking wins and performance
win_history = []                          # 1 = win, 0 = lose
hsize = qmaze.maze.size // 2              # window size for win-rate
win_rate = 0.0

# Just to avoid "variable referenced before assignment" in the print
loss = 0.0
n_episodes = 0

# ============= TRAINING LOOP =============
for epoch in range(n_epoch):

    # Pick a random free starting cell for the pirate
    pirate_cell = random.choice(qmaze.free_cells)

    # Reset the maze to that starting position
    qmaze.reset(pirate_cell)

    # Observe the initial environment state
    envstate = qmaze.observe()
    game_over = False

    # Run one full episode (until win or lose)
    while not game_over:
        # Get valid actions from the current position
        valid_actions = qmaze.valid_actions()
        if not valid_actions:
            # No valid moves, stop this episode
            break

        prev_envstate = envstate

        # ----- ε-greedy action selection -----
        # With probability epsilon: explore (random action)
        # Otherwise: exploit (use model / experience to pick best action)
        if np.random.rand() < epsilon:
            action = random.choice(valid_actions)
        else:
            q_values = experience.predict(prev_envstate)   # GameExperience uses model to predict
            action = np.argmax(q_values[0])

        # Take the action in the environment
        envstate, reward, game_status = qmaze.act(action)

        # Check outcome and record win/lose
        if game_status == 'win':
            win_history.append(1)
            game_over = True
        elif game_status == 'lose':
            win_history.append(0)
            game_over = True
        else:
            game_over = False

        # Store this step in replay memory
        episode = [prev_envstate, action, reward, envstate, game_over]
        experience.remember(episode)

        # Count episodes (steps) for reporting
        n_episodes += 1

        # ----- Train the neural network using a batch from memory -----
        inputs, targets = experience.get_data(data_size=data_size)
        if inputs is not None and len(inputs) > 0:
            history = model.fit(
                inputs,
                targets,
                epochs=1,
                batch_size=16,
                verbose=0,
            )
            # Last batch loss
            loss = history.history['loss'][-1]

    # Periodically update the target network to follow the main model
    if epoch % target_update_freq == 0:
        target_model.set_weights(model.get_weights())

    # ----- Reporting & stopping logic (this part was already in your starter) -----
    win_rate = sum(win_history[-hsize:]) / hsize if len(win_history) >= hsize else 0.0

    # Print the epoch, loss, episode, win count, win rate, and time for each epoch
    dt = datetime.datetime.now() - start_time
    t = format_time(dt.total_seconds())
    print(
        "Epoch: {:03d}/{:d} | Loss: {:.4f} | Episodes: {:d} | "
        "Win count: {:d} | Win rate: {:.3f} | time: {}".format(
            epoch, n_epoch - 1, loss, n_episodes, sum(win_history), win_rate, t
        )
    )

    # Adjust epsilon (exploration) over time:
    # if we're doing well, lock it in lower; otherwise decay slowly
    if win_rate > 0.9:
        epsilon = 0.05
    else:
        epsilon = max(epsilon * epsilon_decay, epsilon_min)

    # If we’re basically perfect and pass the completion check, stop early
    if win_rate >= 0.999 and completion_check(model, maze):
        print(f"Reached 100% win rate at epoch {epoch}")
        break

total_time = format_time((datetime.datetime.now() - start_time).total_seconds())
print("Training complete in:", total_time)

def format_time(seconds): if seconds < 400: s = float(seconds) return "%.1f seconds" % (s,) elif seconds < 4000: m = seconds / 60.0 return "%.2f minutes" % (m,) else: h = seconds / 3600.0 return "%.2f hours" % (h,)

def format_time(seconds):
    """Convert seconds to a nicely formatted string."""
    if seconds < 60:
        return f"{seconds:.1f} seconds"
    minutes = seconds / 60
    if minutes < 60:
        return f"{minutes:.1f} minutes"
    hours = minutes / 60
    return f"{hours:.1f} hours"

def qtrain(model, maze, **opt):
    import random
    import datetime
    from collections import deque
    import numpy as np

    n_epoch           = opt.get('n_epoch', 3000)
    max_memory        = opt.get('max_memory', maze.size * 8)
    data_size         = opt.get('data_size', 32)
    gamma             = opt.get('gamma', 0.95)

    epsilon           = opt.get('epsilon', 1.0)
    epsilon_min       = opt.get('epsilon_min', 0.05)
    epsilon_decay     = opt.get('epsilon_decay', 0.995)

    qmaze = TreasureMaze(maze)
    pirate_cell = (0, 0)  # safe default start for this maze

    memory = deque(maxlen=max_memory)
    win_history = []

    start_time = datetime.datetime.now()
    max_steps = qmaze.maze.size * 4  # matches play_game()

    for epoch in range(n_epoch):
        qmaze.reset(pirate_cell)
        envstate = qmaze.observe()

        episode_loss = 0.0
        game_status = 'not_over'

        for step in range(max_steps):
            valid_actions = qmaze.valid_actions()

            if random.random() < epsilon:
                action = random.choice(valid_actions)
            else:
                q_values = model.predict(envstate, verbose=0)[0]

                # mask invalid actions so we never pick them
                masked_q = np.full_like(q_values, -1e9, dtype=np.float32)
                for a in valid_actions:
                    masked_q[a] = q_values[a]

                action = int(np.argmax(masked_q))


            # IMPORTANT: act returns (envstate, reward, game_status)
            next_envstate, reward, game_status = qmaze.act(action)

            done = (game_status == 'win') or (game_status == 'lose')

            # store transition
            memory.append((envstate, action, reward, next_envstate, done))

            envstate = next_envstate

            # train from replay
            if len(memory) >= data_size:
                batch = random.sample(memory, data_size)

                states = np.vstack([b[0] for b in batch])
                next_states = np.vstack([b[3] for b in batch])

                q_targets = model(states, training=False).numpy()
                q_next = model(next_states, training=False).numpy()


                for i, (s, a, r, ns, d) in enumerate(batch):
                    if d:
                        q_targets[i][a] = r
                    else:
                        q_targets[i][a] = r + gamma * np.max(q_next[i])

                history = model.fit(states, q_targets, epochs=1, verbose=0)
                episode_loss += float(history.history['loss'][0])

            if done:
                break

        # decay epsilon
        if epsilon > epsilon_min:
            epsilon = max(epsilon_min, epsilon * epsilon_decay)

        win = 1 if game_status == 'win' else 0
        win_history.append(win)

        recent = win_history[-50:]
        win_rate = sum(recent) / len(recent)

        dt = datetime.datetime.now() - start_time
        t = format_time(dt.total_seconds())

        print("Epoch: {:03d}/{:d} | Loss: {:.4f} | Win count: {:d} | Win rate: {:.3f} | epsilon: {:.3f} | time: {}".format(
            epoch, n_epoch-1, episode_loss, sum(win_history), win_rate, epsilon, t))

        if len(win_history) >= 50 and win_rate >= 0.9:
            print("Stopping early: win rate >= 0.9 over last 50 episodes.")
            break

    total_time = datetime.datetime.now() - start_time
    print("Training complete in:", format_time(total_time.total_seconds()))
    return model, win_history
